cmake_minimum_required(VERSION 3.20)
project(mamba3_cpu_decode LANGUAGES C CXX)

if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CONFIG} "${CMAKE_BINARY_DIR}/bin")
endforeach()
if(WIN32)
  add_compile_definitions(_WIN32_WINNT=0x0A00 NOMINMAX)
endif()

find_package(OpenMP QUIET)
enable_testing()
add_subdirectory(src)

add_executable(machine src/machine.cpp)
target_compile_features(machine PRIVATE cxx_std_17)
if(WIN32)
  target_link_libraries(machine PRIVATE psapi)
endif()

add_executable(stream vendor/stream.c)
set(PROJECT_STREAM_ARRAY_SIZE 20000000 CACHE STRING "STREAM elements per array")
target_compile_definitions(stream PRIVATE STREAM_ARRAY_SIZE=${PROJECT_STREAM_ARRAY_SIZE} NTIMES=20)
if(OpenMP_C_FOUND)
  target_link_libraries(stream PRIVATE OpenMP::OpenMP_C)
endif()
if(MSVC)
  target_include_directories(stream PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/compat/msvc")
endif()
