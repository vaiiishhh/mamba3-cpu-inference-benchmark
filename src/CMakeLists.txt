add_executable(mamba_decode main.cpp mixer.cpp step_scalar.cpp step_simd.cpp)
target_compile_features(mamba_decode PRIVATE cxx_std_17)
if(OpenMP_CXX_FOUND)
  target_link_libraries(mamba_decode PRIVATE OpenMP::OpenMP_CXX)
endif()
if(WIN32)
  target_link_libraries(mamba_decode PRIVATE psapi)
endif()
if(MSVC)
  set_source_files_properties(step_scalar.cpp PROPERTIES COMPILE_OPTIONS "/Qvec-;/fp:strict")
  set_source_files_properties(mixer.cpp PROPERTIES COMPILE_OPTIONS "/fp:strict")
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|x86_64)$")
    option(MAMBA_MSVC_AVX2 "Build the SIMD translation unit for AVX2/FMA CPUs" OFF)
    if(MAMBA_MSVC_AVX2)
      set_source_files_properties(step_simd.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
    endif()
  endif()
else()
  # NEON is part of AArch64. Avoid host CPU probing when targeting arm64
  # from an x86-64 build tool running under Rosetta.
  if(NOT CMAKE_OSX_ARCHITECTURES AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    target_compile_options(mamba_decode PRIVATE -march=native)
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set_source_files_properties(step_scalar.cpp PROPERTIES COMPILE_OPTIONS
      "-fno-vectorize;-fno-slp-vectorize;-ffp-contract=off")
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    set_source_files_properties(step_scalar.cpp PROPERTIES COMPILE_OPTIONS
      "-fno-tree-vectorize;-ffp-contract=off")
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    set_source_files_properties(mixer.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
  endif()
endif()
add_test(NAME mamba_smoke COMMAND mamba_decode bench
  --variant siso --backend scalar --threads 1 --seconds 0.01)
